## 六、关键实现细节

### 6.1 每日金句刷新
```javascript
// 从金句库随机抽取，排除当前显示的金句
function getRandomQuote(excludeId) {
  const quotes = db.collection('quotes').get()
  const filtered = quotes.filter(q => q.id !== excludeId)
  return filtered[Math.floor(Math.random() * filtered.length)]
}
```

### 6.2 迷你播放器显示逻辑
```javascript
// 有播放内容且正在播放时显示
showMiniPlayer() {
  return this.globalData.player.isPlaying && this.globalData.player.currentBook
}
```

### 6.3 分类筛选
```javascript
// 书籍列表页分类筛选
onCategoryChange(e) {
  const categoryId = e.currentTarget.dataset.id
  this.setData({ categoryId })
  this.loadBooks(categoryId)
}
```
